        function renderChecklist() {

            const container = document.getElementById('checklist');

            container.innerHTML = '';

            let globalTaskIndex = 0;

            data.forEach((sec, secIdx) => {

                const card = document.createElement('div');

                card.className = 'card';

               

                let html = `<div class="section-title">${sec.category}</div><div class="task-list">`;

                sec.tasks.forEach((task) => {

                    const id = `task_${globalTaskIndex}`;

                    const workerId = `worker_${globalTaskIndex}`;

                   

                    html += `

                        <div class="task-item" id="item_${id}">

                            <div class="task-main">

                                <input type="checkbox" id="${id}" onchange="toggleTask('${id}', '${workerId}')">

                                <label for="${id}" class="task-label">${task}</label>

                            </div>

                            <input type="text" id="${workerId}" class="worker-input" placeholder="Όνομα υπαλλήλου..." oninput="saveData()">

                        </div>

                    `;

                    globalTaskIndex++;

                });

                html += `</div>`;

                card.innerHTML = html;

                container.appendChild(card);

            });

            // Set current date by default if empty

            if(!document.getElementById('dateInfo').value) {

                document.getElementById('dateInfo').valueAsDate = new Date();

            }

            loadData();

        }

        function toggleTask(taskId, workerId) {

            const checkbox = document.getElementById(taskId);

            const item = document.getElementById(`item_${taskId}`);

           

            if (checkbox.checked) {

                item.classList.add('completed');

            } else {

                item.classList.remove('completed');

            }

           

            updateProgress();

            saveData();

        }

        function updateProgress() {

            const checkboxes = document.querySelectorAll('input[type="checkbox"]');

            const total = checkboxes.length;

            let checked = 0;

            checkboxes.forEach(cb => {

                if (cb.checked) checked++;

            });

            const percentage = total > 0 ? Math.round((checked / total) * 100) : 0;

            document.getElementById('progressBar').style.width = percentage + '%';

            document.getElementById('progressText').innerText = percentage + '%';

        }

        function saveData() {

            const state = {

                carInfo: document.getElementById('carInfo').value,

                dateInfo: document.getElementById('dateInfo').value,

                tasks: {},

                workers: {}

            };

            const checkboxes = document.querySelectorAll('input[type="checkbox"]');

            checkboxes.forEach(cb => {

                state.tasks[cb.id] = cb.checked;

            });

            const workers = document.querySelectorAll('.worker-input');

            workers.forEach(w => {

                state.workers[w.id] = w.value;

            });

            localStorage.setItem('detailing_app_data', JSON.stringify(state));

            updateProgress();

        }

        function loadData() {

            const saved = localStorage.getItem('detailing_app_data');

            if (!saved) return;

            const state = JSON.parse(saved);

            document.getElementById('carInfo').value = state.carInfo || '';

            if (state.dateInfo) document.getElementById('dateInfo').value = state.dateInfo;

            if (state.tasks) {

                Object.keys(state.tasks).forEach(id => {

                    const cb = document.getElementById(id);

                    if (cb) {

                        cb.checked = state.tasks[id];

                        const item = document.getElementById(`item_${id}`);

                        if (cb.checked && item) item.classList.add('completed');

                    }

                });

            }

            if (state.workers) {

                Object.keys(state.workers).forEach(id => {

                    const input = document.getElementById(id);

                    if (input) input.value = state.workers[id];

                });

            }

            updateProgress();

        }

        function resetForm() {

            if (confirm("Θέλετε να καθαρίσετε τη φόρμα για νέο αμάξι;")) {

                localStorage.removeItem('detailing_app_data');

                document.getElementById('carInfo').value = '';

                document.getElementById('dateInfo').valueAsDate = new Date();

               

                document.querySelectorAll('input[type="checkbox"]').forEach(cb => {

                    cb.checked = false;

                });

               

                document.querySelectorAll('.task-item').forEach(item => {

                    item.classList.remove('completed');

                });

                document.querySelectorAll('.worker-input').forEach(w => {

                    w.value = '';

                });

                updateProgress();

            }

        }
